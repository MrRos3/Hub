        BorderSizePixel = 0,
        Text = isFavorite(entry.Id) and "♥" or "♡",
        TextColor3 = isFavorite(entry.Id) and THEME.AccentBright or Color3.fromHex("#666671"),
        TextSize = 16,
        FontFace = font(Enum.FontWeight.Medium),
        AutoButtonColor = false,
        ZIndex = zIndex or 5,
        Parent = parentFrame,
    }, { corner(5) })

    button.MouseEnter:Connect(function()
        tween(button, 0.12, { BackgroundTransparency = 0.25, TextColor3 = THEME.AccentBright })
    end)
    button.MouseLeave:Connect(function()
        tween(button, 0.12, {
            BackgroundTransparency = 1,
            TextColor3 = isFavorite(entry.Id) and THEME.AccentBright or Color3.fromHex("#666671"),
        })
    end)
    button.MouseButton1Click:Connect(function()
        setFavorite(entry.Id, not isFavorite(entry.Id))
        renderContent()
    end)
    return button
end

local function makeLoadButton(parentFrame, entry, size, position, zIndex)
    local button = create("TextButton", {
        Position = position,
        Size = size,
        BackgroundColor3 = THEME.Accent,
        BorderSizePixel = 0,
        Text = "",
        AutoButtonColor = false,
        ZIndex = zIndex or 6,
        Parent = parentFrame,
    }, { corner(5) })

    local label = create("TextLabel", {
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        Text = loading[entry.Id] and "◌  Loading..." or loaded[entry.Id] and "✓  Loaded" or "▶  Load Script",
        TextColor3 = Color3.fromHex("#E7DFF6"),
        TextSize = size.Y.Offset >= 34 and 10 or 9,
        FontFace = font(Enum.FontWeight.SemiBold),
        ZIndex = (zIndex or 6) + 1,
        Parent = button,
    })

    if loading[entry.Id] then
        local spinner = create("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromOffset(10, math.floor(size.Y.Offset / 2)),
            Size = UDim2.fromOffset(10, 10),
            BackgroundTransparency = 1,
            ZIndex = (zIndex or 6) + 2,
            Parent = button,
        }, { corner(99), stroke(Color3.fromHex("#E7DFF6"), 0.25, 1) })
        local cut = create("Frame", {
            Position = UDim2.fromOffset(5, -1),
            Size = UDim2.fromOffset(6, 5),
            BackgroundColor3 = THEME.Accent,
            BorderSizePixel = 0,
            ZIndex = (zIndex or 6) + 3,
            Parent = spinner,
        })
        task.spawn(function()
            while spinner and spinner.Parent and loading[entry.Id] do
                spinner.Rotation = 0
                local a = TweenService:Create(spinner, TweenInfo.new(0.7, Enum.EasingStyle.Linear), { Rotation = 360 })
                a:Play()
                a.Completed:Wait()
            end
        end)
    end

    button.MouseEnter:Connect(function()
        if not loading[entry.Id] then
            tween(button, 0.14, { BackgroundColor3 = THEME.AccentHover })
        end
    end)
    button.MouseLeave:Connect(function()
        tween(button, 0.14, { BackgroundColor3 = THEME.Accent })
    end)
    button.MouseButton1Click:Connect(function()
        runRemote(entry)
    end)
    return button
end
