    end

    local visible = collectVisible()
    if #visible == 0 then
        layout.CellSize = UDim2.new(1, -2, 0, 220)
        layout.CellPadding = UDim2.fromOffset(0, 0)
        layout.FillDirectionMaxCells = 1
        local empty = createEmptyState()
        empty.Parent = content
    else
        for index, entry in ipairs(visible) do
            local card = currentView == "grid" and createGridCard(entry, index) or createListCard(entry, index)
            card.Parent = content
            table.insert(cardObjects, card)
        end
    end

    task.defer(updateCanvas)
end

-- Details -------------------------------------------------------------------
local function closeDetails()
    selectedEntry = nil
    if detailBackdrop and detailBackdrop.Parent then
        detailBackdrop:Destroy()
    end
    if detailPanel and detailPanel.Parent then
        local panel = detailPanel
        tween(panel, 0.2, { Position = UDim2.new(1, 0, 0, 0) })
        task.delay(0.21, function()
            if panel and panel.Parent then
                panel:Destroy()
            end
        end)
    end
    detailBackdrop = nil
    detailPanel = nil
end
