local function collectVisible()
    local query = normalize(searchBox.Text)
    local result = {}
    for _, entry in ipairs(SCRIPTS) do
        local matchesQuery = query == ""
            or contains(entry.Name, query)
            or contains(entry.Game, query)
            or contains(entry.Category, query)
            or contains(entry.Description, query)
        if not matchesQuery then
            for _, tag in ipairs(entry.Tags or {}) do
                if contains(tag, query) then
                    matchesQuery = true
                    break
                end
            end
        end

        local matchesFilter = currentFilter == "All Scripts"
            or (currentFilter == "Games" and entry.Category == "GAME SCRIPT")
            or (currentFilter == "Universal" and entry.Category == "UNIVERSAL")
            or (currentFilter == "Favorites" and isFavorite(entry.Id))

        if matchesQuery and matchesFilter then
            table.insert(result, entry)
        end
    end

    if currentSort == "Alphabetical" then
        table.sort(result, function(a, b)
            return a.Name:lower() < b.Name:lower()
        end)
    elseif currentSort == "Most Used" then
        table.sort(result, function(a, b)
            return (a.UsageCount or 0) > (b.UsageCount or 0)
        end)
