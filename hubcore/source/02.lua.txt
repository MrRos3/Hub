local defaultSettings = {
    favorites = { "velora-piano" },
    view = "grid",
    sort = "Recently Updated",
}

local settingsData = {
    favorites = {},
    view = defaultSettings.view,
    sort = defaultSettings.sort,
}

local function copyDefaults()
    settingsData.favorites = {}
    for _, id in ipairs(defaultSettings.favorites) do
        settingsData.favorites[id] = true
    end
    settingsData.view = defaultSettings.view
    settingsData.sort = defaultSettings.sort
end

copyDefaults()

if canCache and type(readfile) == "function" and isfile(SETTINGS_PATH) then
    local ok, decoded = pcall(function()
        return HttpService:JSONDecode(readfile(SETTINGS_PATH))
    end)
    if ok and type(decoded) == "table" then
        settingsData.favorites = {}
        if type(decoded.favorites) == "table" then
            for _, id in ipairs(decoded.favorites) do
                settingsData.favorites[tostring(id)] = true
            end
        end
        if decoded.view == "grid" or decoded.view == "list" then
            settingsData.view = decoded.view
        end
        if decoded.sort == "Recently Updated" or decoded.sort == "Alphabetical" or decoded.sort == "Most Used" then
            settingsData.sort = decoded.sort
        end
    end
end

local function saveSettings()
    if not canCache then
        return
    end
    local favorites = {}
    for id, enabled in pairs(settingsData.favorites) do
        if enabled then
            table.insert(favorites, id)
        end
    end
    table.sort(favorites)
    local payload = {
        favorites = favorites,
        view = settingsData.view,
        sort = settingsData.sort,
    }
    pcall(function()
        writefile(SETTINGS_PATH, HttpService:JSONEncode(payload))
    end)
end

    local ok, result = pcall(function()

            return gethui()

        return result

    return player:WaitForChild("PlayerGui")
end
