    else
        table.sort(result, function(a, b)
            return (a.UpdatedAt or 0) > (b.UpdatedAt or 0)
        end)
    end

    return result
end

local renderContent
local openDetails

local function runRemote(entry)
    if loading[entry.Id] then
        return
    end
    loading[entry.Id] = true
    loaded[entry.Id] = nil
    if renderContent then
        renderContent()
    end
    if selectedEntry and selectedEntry.Id == entry.Id and openDetails then
        task.defer(function() openDetails(entry) end)
    end

    task.spawn(function()
        local started = os.clock()
        local ok, err = pcall(function()
            if type(loadstring) ~= "function" then
                error("loadstring is unavailable in this environment")
            end
            local sep = string.find(entry.Url, "?", 1, true) and "&" or "?"
            local cache = tostring(os.time()) .. "-" .. tostring(math.random(100000, 999999))
            local source = game:HttpGet(entry.Url .. sep .. "v=" .. cache)
            if type(source) ~= "string" or source == "" then
                error("download returned no source")
            end
            local chunk, compileError = loadstring(source)
            if not chunk then
                error(compileError or "compile failed")
            end
            chunk()
        end)

        local remaining = 0.55 - (os.clock() - started)
        if remaining > 0 then
            task.wait(remaining)
        end

        loading[entry.Id] = nil
        if ok then
            loaded[entry.Id] = true
            showToast(entry.Name .. " loaded", "Ready to use in your Roblox session.", "success")
            task.delay(1.6, function()
                loaded[entry.Id] = nil
                if renderContent then
                    renderContent()
                end
            end)
        else
            showToast(entry.Name .. " could not load", truncate(err, 60), "error")
        end

        if renderContent then
            renderContent()
        end
        if selectedEntry and selectedEntry.Id == entry.Id and openDetails then
            task.defer(function() openDetails(entry) end)
        end
    end)
end

local function styleFilterButtons()
    for name, button in pairs(filterButtons) do
        local active = name == currentFilter
        button.BackgroundTransparency = active and 0 or 1
        button.BackgroundColor3 = THEME.AccentSoft
        button.TextColor3 = active and Color3.fromHex("#D9CFF1") or THEME.Muted
        local s = button:FindFirstChildOfClass("UIStroke")
        if s then
            s.Transparency = active and 0.15 or 1
            s.Color = THEME.AccentBorder
        end
        if name == "Favorites" then
            button.Text = "Favorites  " .. tostring(favoriteCount())
        else
            button.Text = name
        end
    end
end

    local gridActive = currentView == "grid"

        end

end
