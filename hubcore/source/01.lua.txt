local function font(weight)
    local ok, face = pcall(function()
        return Font.new(FONT_ID, weight or Enum.FontWeight.Regular, Enum.FontStyle.Normal)
    end)
    if ok and face then
        return face
    end
    return Font.fromEnum(Enum.Font.Gotham)
end

local function create(className, props, children)
    local object = Instance.new(className)
    for key, value in pairs(props or {}) do
        object[key] = value
    end
    for _, child in ipairs(children or {}) do
        child.Parent = object
    end
    return object
end

local function corner(radius)
    return create("UICorner", { CornerRadius = UDim.new(0, radius) })
end

local function stroke(color, transparency, thickness)
    return create("UIStroke", {
        Color = color or THEME.Border,
        Transparency = transparency or 0,
        Thickness = thickness or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    })
end

local function tween(object, duration, props, style, direction)
    local anim = TweenService:Create(
        object,
        TweenInfo.new(
            duration or 0.2,
            style or Enum.EasingStyle.Quint,
            direction or Enum.EasingDirection.Out
        ),
        props
    )
    anim:Play()
    return anim
end

local function normalize(value)
    return string.lower(tostring(value or ""))
end

local function contains(value, query)
    return query == "" or string.find(normalize(value), query, 1, true) ~= nil
end

local function truncate(value, maxLength)
    value = tostring(value or "")
    if #value <= maxLength then
        return value
    end
    return value:sub(1, math.max(1, maxLength - 1)) .. "…"
end

-- Optional executor-backed local asset cache. The UI still works when unavailable.
local customAsset = getcustomasset or getsynasset
local canCache = type(writefile) == "function"
    and type(isfile) == "function"
    and type(makefolder) == "function"
    and type(isfolder) == "function"
    and type(customAsset) == "function"

local function ensureFolder(path)
    if not canCache then
        return false
    end
    if isfolder(path) then
        return true
    end
    return pcall(makefolder, path)
end

if canCache then
    ensureFolder(CACHE_ROOT)
    ensureFolder(CACHE_ROOT .. "/assets")
end

local BASE64_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local BASE64_LOOKUP = {}
for i = 1, #BASE64_ALPHABET do
    BASE64_LOOKUP[BASE64_ALPHABET:sub(i, i)] = i - 1
end

local function decodeBase64(data)
    data = tostring(data or ""):gsub("%s+", "")
    local out = {}
    local outCount = 0

    for i = 1, #data, 4 do
        local c1 = data:sub(i, i)
        local c2 = data:sub(i + 1, i + 1)
        local c3 = data:sub(i + 2, i + 2)
        local c4 = data:sub(i + 3, i + 3)
        local a = BASE64_LOOKUP[c1]
        local b = BASE64_LOOKUP[c2]
        if a == nil or b == nil then
            break
        end
        local c = c3 == "=" and 0 or BASE64_LOOKUP[c3]
        local d = c4 == "=" and 0 or BASE64_LOOKUP[c4]
        if c == nil or d == nil then
            break
        end

        local n = a * 262144 + b * 4096 + c * 64 + d
        outCount += 1
        out[outCount] = string.char(math.floor(n / 65536) % 256)
        if c3 ~= "=" then
            outCount += 1
            out[outCount] = string.char(math.floor(n / 256) % 256)
        end
        if c4 ~= "=" then
            outCount += 1
            out[outCount] = string.char(n % 256)
        end
    end

    return table.concat(out)
end

local function cacheRemoteAsset(url, path)
    if not canCache or not url or url == "" then
        return ""
    end
    if not isfile(path) then
        local ok, bytes = pcall(function()
            return game:HttpGet(url)
        end)
        if ok and type(bytes) == "string" then
            if string.find(url, ".b64.txt", 1, true) then
                bytes = decodeBase64(bytes)
            end
            if #bytes > 100 then
                pcall(writefile, path, bytes)
            end
        end
    end
    if isfile(path) then
        local ok, asset = pcall(customAsset, path)
        if ok and type(asset) == "string" then
            return asset
        end
    end
    return ""
end

local function setRemoteImage(imageLabel, url, cacheName)
    if not imageLabel or not imageLabel.Parent or not url or url == "" then
        return
    end
    task.spawn(function()
        local asset = cacheRemoteAsset(url, CACHE_ROOT .. "/assets/" .. cacheName)
        if asset ~= "" and imageLabel and imageLabel.Parent then
            imageLabel.Image = asset
            imageLabel.ImageTransparency = 0
        end
    end)
end
