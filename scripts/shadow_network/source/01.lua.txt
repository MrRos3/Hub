if ENV.__SALTY_TSN_VANTA then
    local previous = ENV.__SALTY_TSN_VANTA

    -- Ask the old build to stop normally first.
    if previous.Stop then
        pcall(function()
            previous:Stop()
        end)
    end

    -- v1.3 briefly wrapped Window.Destroy itself. If that old wrapper failed
    -- to remove the UI, bypass it and call VantaUI's native close object.
    if previous.Window
        and previous.Window.Destroyed ~= true
        and type(previous.Window.Close) == "function"
    then
        pcall(function()
            local closeObject = previous.Window:Close()
            if closeObject and type(closeObject.Destroy) == "function" then
                closeObject:Destroy()
            end
        end)
    end

    ENV.__SALTY_TSN_VANTA = nil
end

-- These modules belong to the lobby flow and are not guaranteed to exist
-- inside the gameplay place. Never use an unbounded WaitForChild here,
-- because that would freeze Salty before VantaUI is created.
local function safeRequireChild(parent, name)
    if not parent then
        return nil
    end

    local moduleScript = parent:FindFirstChild(name)

    if not moduleScript or not moduleScript:IsA("ModuleScript") then
        return nil
    end

    local ok, result = pcall(require, moduleScript)

    if ok then
        return result
    end

    return nil
end

end

local function notify(title, content, icon)
    if State.Unloading then
        return
    end

    pcall(function()
        VantaUI:Notify({
            Title = title or "Salty",
            Content = tostring(content or ""),
            Icon = icon or "sparkles",
        })
    end)
end

local function getSpeedModule()
    if State.SpeedModule then
        return State.SpeedModule
    end

    local modules = ReplicatedStorage:FindFirstChild("Modules")
    local systems = modules and modules:FindFirstChild("Systems")
    local scriptObject = systems and systems:FindFirstChild("SetWalkSpeed")

    if not scriptObject or not scriptObject:IsA("ModuleScript") then
        return nil
    end

    local ok, module = pcall(require, scriptObject)
    if ok and type(module) == "table" then
        State.SpeedModule = module
        return module
    end

    return nil
end

local SPEED_PASSIVE = {
    Name = "SaltySpeed",
    Percent = 0,
    EndurancePercent = 0,
}

local function classifyStance(module, hum)
    if not module or not hum then
        return "walk"
    end

    if hum.WalkSpeed <= 0.05 then
        return "stopped"
    end

    local candidates = {}

    local function add(name, fn)
        if type(fn) == "function" then
            local ok, value = pcall(fn, LocalPlayer)
            if ok and type(value) == "number" then
                candidates[#candidates + 1] = { Name = name, Speed = value }
            end
        end
    end

    add("crouch", module.GiveCrouchSpeed)
    add("walk", module.GiveWalkSpeed)
    add("run", module.GiveRunSpeed)

    local bestName = "walk"
    local bestDistance = math.huge

    for _, item in ipairs(candidates) do
        local d = math.abs(hum.WalkSpeed - item.Speed)
        if d < bestDistance then
            bestDistance = d
            bestName = item.Name
        end
    end

    return bestName
end

local function setStanceSpeed(module, hum, stance)
    if not module or not hum or stance == "stopped" then
        return
    end

    local fn
    if stance == "crouch" then
        fn = module.GiveCrouchSpeed
    elseif stance == "run" then
        fn = module.GiveRunSpeed
    else
        fn = module.GiveWalkSpeed
    end

    if type(fn) == "function" then
        local ok, value = pcall(fn, LocalPlayer)
        if ok and type(value) == "number" then
            hum.WalkSpeed = value
        end
    end
end

local function applySpeedAssist(showNotice)
    local module = getSpeedModule()
    local hum = humanoid()

    if not module or not hum then
        if showNotice then
            notify("Movement", "The game's SetWalkSpeed module or Humanoid is not ready.", "circle-x")
        end
        return false
    end

    -- Touch GiveWalkSpeed once so the game's private passive table is initialized.
    pcall(function()
        module.GiveWalkSpeed(LocalPlayer)
    end)

    local stance = classifyStance(module, hum)

    if type(module.RemovePassive) == "function" then
        pcall(function()
            module.RemovePassive(LocalPlayer, SPEED_PASSIVE)
        end)
    end

    if State.SpeedEnabled and type(module.AddPassive) == "function" then
        SPEED_PASSIVE.Percent = math.max(0, (State.SpeedMultiplier - 1) * 100)
        pcall(function()
            module.AddPassive(LocalPlayer, SPEED_PASSIVE)
        end)
    end

    setStanceSpeed(module, hum, stance)

    if showNotice then
        if State.SpeedEnabled then
            notify(
                "Movement",
                ("Game-aware speed multiplier set to %.2fx."):format(State.SpeedMultiplier),
                "gauge"
            )
        else
            notify("Movement", "Speed multiplier removed.", "gauge")
        end
    end

    return true
end

local function restoreCollisions()
    for part, value in pairs(State.OriginalCollisions) do
        if part and part.Parent then
            pcall(function()
                part.CanCollide = value
            end)
        end
    end
    table.clear(State.OriginalCollisions)
end

local function applyNoclip()
    local char = character()
    if not char then
        return
    end

    for _, object in ipairs(char:GetDescendants()) do
        if object:IsA("BasePart") then
            if State.OriginalCollisions[object] == nil then
                State.OriginalCollisions[object] = object.CanCollide
            end
            object.CanCollide = false
        end
    end
end

local function saveLighting()
    if State.OriginalLighting then
        return
    end

    State.OriginalLighting = {
        Brightness = Lighting.Brightness,
        ClockTime = Lighting.ClockTime,
        FogEnd = Lighting.FogEnd,
        GlobalShadows = Lighting.GlobalShadows,
        Ambient = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        ExposureCompensation = Lighting.ExposureCompensation,
    }
end

local function applyFullbright()
    saveLighting()
    Lighting.Brightness = 2.5
    Lighting.ClockTime = 14
    Lighting.FogEnd = 1000000
    Lighting.GlobalShadows = false
