State.SelectedRunItem = InitialItemValues[1]

local function refreshRunItemsUi(showNotice)
    local previous = selectedRunItemRecord(State.SelectedRunItem)
    local previousKey = previous and previous.Key or nil

    ItemsModule = nil
    local values = dropdownValuesForRunItems()
    local selected = values[1]

    if previousKey then
        for _, value in ipairs(values) do
            if value.Key == previousKey then
                selected = value
                break
            end
        end
    end

    State.SelectedRunItem = selected

    if ItemsDropdown then
        ItemsDropdown:Refresh(values)
        if selected then
            ItemsDropdown:Select(selected)
        end
    end

    ItemsParagraph:SetDesc(runItemsText(selected))

    if showNotice then
        notify("Items", ("Loaded %d item key(s) from the game."):format(#values), "refresh-cw")
    end
end
