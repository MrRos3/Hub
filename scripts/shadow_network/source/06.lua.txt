
    return model:FindFirstChild("Head")
        or model:FindFirstChild("HumanoidRootPart")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart")
end

local function looksLikeThreat(model)
    if not model or not model:IsA("Model") then
        return false
    end

    if playerFromCharacterModel(model) then
        return false
    end

    if isKnownThreatName(model.Name) then
        return modelAnchor(model) ~= nil
    end

    local parent = model.Parent
    if parent and (
        isKnownThreatName(parent.Name)
        or normalized(parent.Name):find("monster", 1, true)
        or normalized(parent.Name):find("npc", 1, true)
    ) then
        return modelAnchor(model) ~= nil
    end

    for name, value in pairs(model:GetAttributes()) do
        local key = normalized(name)
        if (key:find("entity", 1, true) or key:find("hostile", 1, true))
            and value ~= false and value ~= 0 and value ~= nil
        then
            return modelAnchor(model) ~= nil
        end
    end

    if State.UnknownNPCs and model:FindFirstChildOfClass("Humanoid") then
        return modelAnchor(model) ~= nil
    end

    return false
end

local function createEspEntry(kind, model, labelName)
    local anchor = modelAnchor(model)
    if not anchor then
        return nil
    end

    if State.EspEntries[model] then
        return State.EspEntries[model]
    end

    local highlight = Instance.new("Highlight")
    highlight.Name = "Salty_" .. kind
    highlight.Adornee = model
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.FillTransparency = kind == "Threat" and 0.72 or 0.82
    highlight.OutlineTransparency = 0

    if kind == "Threat" then
        highlight.FillColor = Color3.fromRGB(255, 70, 70)
        highlight.OutlineColor = Color3.fromRGB(255, 210, 210)
    else
        highlight.FillColor = Color3.fromRGB(85, 180, 255)
        highlight.OutlineColor = Color3.fromRGB(210, 235, 255)
    end

    highlight.Parent = EspFolder

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "Salty_Label"
    billboard.Adornee = anchor
    billboard.AlwaysOnTop = true
    billboard.Size = UDim2.fromOffset(210, 32)
    billboard.StudsOffset = Vector3.new(0, 3.2, 0)
    billboard.Parent = EspFolder

    local text = Instance.new("TextLabel")
    text.BackgroundTransparency = 1
    text.Size = UDim2.fromScale(1, 1)
    text.Font = Enum.Font.GothamMedium
    text.TextSize = 13
    text.TextStrokeTransparency = 0.55
    text.TextColor3 = kind == "Threat" and Color3.fromRGB(255, 120, 120) or Color3.fromRGB(145, 210, 255)
    text.Text = labelName
    text.Parent = billboard

    local entry = {
        Kind = kind,
        Model = model,
        Anchor = anchor,
        Highlight = highlight,
        Billboard = billboard,
        Label = text,
        BaseName = labelName,
    }

    State.EspEntries[model] = entry
    return entry
end

local function destroyEspEntry(model)
    local entry = State.EspEntries[model]
    if not entry then
        return
    end

    pcall(function()
        entry.Highlight:Destroy()
    end)
    pcall(function()
        entry.Billboard:Destroy()
    end)

    State.EspEntries[model] = nil
end

local function clearEsp(kind)
    for model, entry in pairs(State.EspEntries) do
        if not kind or entry.Kind == kind then
            destroyEspEntry(model)
        end
    end
end

local function syncEsp()
    local wanted = {}
    local myRoot = rootPart()

    if State.PlayerESP then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character and player.Character.Parent then
                local model = player.Character
                local anchor = modelAnchor(model)
                if anchor then
                    wanted[model] = true
                    local entry = createEspEntry("Player", model, player.DisplayName or player.Name)
                    if entry then
                        entry.BaseName = player.DisplayName or player.Name
                    end
                end
            end
        end
    end

    if State.ThreatESP then
        for _, object in ipairs(workspace:GetDescendants()) do
            if object:IsA("Model") and looksLikeThreat(object) then
                wanted[object] = true
                createEspEntry("Threat", object, object.Name)
            end
        end
    end

    for model, entry in pairs(State.EspEntries) do
        local valid = wanted[model]
            and entry.Model
            and entry.Model.Parent
            and entry.Anchor
            and entry.Anchor.Parent

        if not valid then
            destroyEspEntry(model)
        elseif myRoot then
            local distance = math.floor((myRoot.Position - entry.Anchor.Position).Magnitude + 0.5)
            local insideRange = distance <= State.EspMaxDistance

            entry.Highlight.Enabled = insideRange
            entry.Billboard.Enabled = insideRange

            if insideRange then
                if State.EspDistance then
                    entry.Label.Text = ("%s • %d studs"):format(entry.BaseName, distance)
                else
                    entry.Label.Text = entry.BaseName
                end
            end
        end
    end
end

local function currentStatusText()
    local level, exp = getLevelInfo()
    local hum = humanoid()

    local lines = {
        "Player Level: " .. tostring(level or "?"),
        "EXP: " .. tostring(exp or "?"),
        "Health: " .. (hum and (math.floor(hum.Health) .. " / " .. math.floor(hum.MaxHealth)) or "?"),
    }

    if IS_GAMEPLAY_PLACE then
        lines[#lines + 1] = gameplayStatusText():gsub("\n", " • ")
    else
        lines[#lines + 1] = "Story: " .. storyProgressText(State.StoryMode):gsub("\n", " • ")
    end

    return table.concat(lines, "\n")
end
