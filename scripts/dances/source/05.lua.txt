    function()

        State.InBattle =

            true


        ResetNotes()


        ------------------------------------------------

        --

        ------------------------------------------------


        BeginAcquireNotes()

    end

)


--==================================================

-- BATTLE END

--==================================================


Runtime:Connect(

    Remotes.BattleEnd.OnClientEvent,

    function()

        State.InBattle =

            false


        ResetNotes()

    end

)


Runtime:Connect(

    Remotes.Cancelled.OnClientEvent,

    function(data)

        local reason =

            type(data) == "table"

            and data.reason

            or nil


        if State.InBattle

            and reason ~= "waiting_song"

        then

            State.InBattle =

                false


            ResetNotes()

        end

    end

)


--==================================================

-- AUTO ACCEPT

--==================================================


Runtime:Connect(

    Remotes.ChallengeInvite.OnClientEvent,

    function()

        if not CONFIG.AutoAcceptChallenges then

            return

        end


        task.delay(

            0.15,

            function()

                if not Runtime.Alive then

                    return

                end


                Remotes.ChallengeReply:FireServer(

                    true

                )

            end

        )

    end

)
