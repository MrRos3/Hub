        ------------------------------------------------


        if untilHit

            > CONFIG.LookAhead

        then

            break

        end


        checked += 1


        local quality =

            ChooseQuality(note)


        ------------------------------------------------

        -- Intentional miss.

        ------------------------------------------------


        if quality == "miss" then

            continue

        end


        local targetTime =

            note.hitTime

            + GetTargetOffset(

                quality

            )

            + CONFIG.TimingOffset


        if now < targetTime then

            continue

        end


        ------------------------------------------------

        -- Already too late.

        ------------------------------------------------


        if now

            > note.hitTime

                + GoodWindow

                - 0.005

        then

            continue

        end


        ------------------------------------------------

        -- HIT BUDGET

        --

        -- This is what removes the huge first-note

        -- frame spike.

        ------------------------------------------------


        if hitsThisHeartbeat

            >= CONFIG.HitBudgetPerHeartbeat

        then

            break

        end


        local ok, err =

            pcall(

                TryHit,

                note.lane

            )


        if ok then

            hitsThisHeartbeat += 1


            State.AutoHits += 1


            if note.hit

                and index

                    == State.Cursor

            then

                State.Cursor =

                    index + 1

            end

        else

            State.Engine =

                "tryHit error: "

                .. tostring(err)


            ------------------------------------------------

            -- Don't repeatedly throw the same failing call

            -- throughout one Heartbeat.

            ------------------------------------------------


            break

        end

    end

end
