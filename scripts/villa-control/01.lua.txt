local function remote(name)

local function safeInvoke(name, ...)
local object = remote(name)
if not object or not object:IsA("RemoteFunction") then
return false, nil, "Missing RemoteFunction: " .. tostring(name)
end
local args = table.pack(...)
local packed = table.pack(pcall(function()
return object:InvokeServer(table.unpack(args, 1, args.n))
end))
if not packed[1] then
return false, nil, tostring(packed[2])
end
return true, packed[2], packed[3], packed[4]
end
local function safeFire(name, ...)
local object = remote(name)
if not object or not object:IsA("RemoteEvent") then
return false, "Missing RemoteEvent: " .. tostring(name)
end
local args = table.pack(...)
local ok, err = pcall(function()
object:FireServer(table.unpack(args, 1, args.n))
end)
return ok, err
end
local function safeDoorFire(name, ...)
local object = DoorRemotes and DoorRemotes:FindFirstChild(name)

return false, "Missing Door RemoteEvent: " .. tostring(name)
end
local args = table.pack(...)
local ok, err = pcall(function()
object:FireServer(table.unpack(args, 1, args.n))
end)
return ok, err
end
local function countTable(value)
if type(value) ~= "table" then
return 0
end
local count = #value
if count > 0 then
return count
end
for _ in pairs(value) do
count += 1
end
return count
end

local function worldHalfHeight(part)
local cf = part.CFrame
local size = part.Size
return math.abs(cf.RightVector.Y) * size.X * 0.5
+ math.abs(cf.UpVector.Y) * size.Y * 0.5
+ math.abs(cf.LookVector.Y) * size.Z * 0.5
end
local function touchPartNaturally(part, holdTime, restoreAfter, fastMode)
if State.MoveBusy then
return false, "movement busy"
end
if not part or not part:IsA("BasePart") or not part.Parent then
return false, "touch part unavailable"
end
local character, humanoid, root = rig()
if not character or not humanoid or not root then
return false, "character not ready"
end
State.MoveBusy = true
local ok, reason = pcall(function()
local origin = character:GetPivot()
humanoid.Sit = false
stopMomentum(root)
local _, yaw, _ = origin:ToOrientation()
local halfHeight = worldHalfHeight(part)
local topY = part.Position.Y + halfHeight
local x = part.Position.X
local z = part.Position.Z
local aboveOffset = fastMode and 2.20 or 3.25
local contactOffset = fastMode and 1.35 or 1.55
local preContactWait = fastMode and 0.015 or 0.10
local above = CFrame.new(x, topY + aboveOffset, z) * CFrame.Angles(0, yaw, 0)
local contact = CFrame.new(x, topY + contactOffset, z) * CFrame.Angles(0, yaw, 0)
character:PivotTo(above)
stopMomentum(root)
if preContactWait > 0 then
task.wait(preContactWait)
end
root.CFrame = contact
root.AssemblyLinearVelocity = Vector3.new(0, fastMode and -18 or -10, 0)
task.wait(holdTime or (fastMode and 0.12 or 0.55))
if restoreAfter and character.Parent then
stopMomentum(root)
character:PivotTo(origin)
stopMomentum(root)
end
end)
State.MoveBusy = false
if not ok then
return false, tostring(reason)
end
return true
end
local function silentPadTouch(part, heartbeatFrames)
if State.MoveBusy then
return false, "movement busy"
end
if not part or not part:IsA("BasePart") or not part.Parent then
return false, "touch target unavailable"
end
local character, humanoid, root = rig()
if not character or not humanoid or not root then
return false, "character not ready"
end
State.MoveBusy = true
local camera = workspace.CurrentCamera
local origin = character:GetPivot()
local oldCameraType = camera and camera.CameraType
local oldCameraSubject = camera and camera.CameraSubject
local oldCameraCFrame = camera and camera.CFrame
local oldAutoRotate = humanoid.AutoRotate
local transparency = {}
for _, object in ipairs(character:GetDescendants()) do
if object:IsA("BasePart") then
transparency[object] = object.LocalTransparencyModifier
end
end
local frames = math.clamp(math.floor(heartbeatFrames or 1), 1, 6)
local ok, err = pcall(function()
humanoid.Sit = false
humanoid.AutoRotate = false
if camera then
camera.CameraType = Enum.CameraType.Scriptable
camera.CFrame = oldCameraCFrame
end
for partObject in pairs(transparency) do
if partObject and partObject.Parent then
partObject.LocalTransparencyModifier = 1
end
end
stopMomentum(root)
local _, yaw, _ = origin:ToOrientation()
local halfHeight = worldHalfHeight(part)
local topY = part.Position.Y + halfHeight
local contact =
CFrame.new(part.Position.X, topY + 1.20, part.Position.Z)
* CFrame.Angles(0, yaw, 0)
character:PivotTo(contact)
stopMomentum(root)
root.AssemblyLinearVelocity = Vector3.new(0, -24, 0)
RunService.Stepped:Wait()
for _ = 1, frames do
RunService.Heartbeat:Wait()
if camera then
camera.CFrame = oldCameraCFrame
end
end
stopMomentum(root)
character:PivotTo(origin)
stopMomentum(root)
if camera then
camera.CFrame = oldCameraCFrame
end
RunService.RenderStepped:Wait()
end)
pcall(function()
humanoid.AutoRotate = oldAutoRotate
end)
for partObject, value in pairs(transparency) do
if partObject and partObject.Parent then
pcall(function()
partObject.LocalTransparencyModifier = value
end)
end
end
if camera then
pcall(function()
camera.CameraType = oldCameraType or Enum.CameraType.Custom
camera.CameraSubject = oldCameraSubject
camera.CFrame = oldCameraCFrame
end)
end
State.MoveBusy = false
if not ok then
return false, tostring(err)
end
return true
end
