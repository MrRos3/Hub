 = false
if State.AutoBuild then
State.AutoBuild = false
if AutoBuildControl then
AutoBuildControl:Set(false, false)
end
end
notify("Salty", "Automations disabled.", "circle-stop")
end,
})
function State:Stop()
self.Running = false
self.AutoCash = false
self.AutoDaily = false
self.AutoBuild = false
pcall(function()
local horn = remote("HornRequest")
if horn and horn:IsA("RemoteEvent") then
horn:FireServer(false)
end
end)
for _, connection in ipairs(self.Connections) do
pcall(function()
connection:Disconnect()
end)
end
table.clear(self.Connections)
if self.Window then
pcall(function()
if self.Window.Destroy then
self.Window:Destroy()
elseif self.Window.Close then
self.Window:Close()
end
end)
end
end

end)
end)
end
