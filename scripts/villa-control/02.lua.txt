r > before then

end
return true
end

local function truthyPaidValue(value)
local kind = typeof(value)
if kind == "boolean" then
return value == true
end
if kind == "number" then
return value > 0
end
if kind == "string" then
local s = string.lower(value)
return s ~= ""
and s ~= "0"
and s ~= "false"
and s ~= "none"
and s ~= "nil"
and s ~= "free"
end
return value ~= nil
end
local function hasPaidAttribute(instance)
for _, name in ipairs(PAID_ATTRIBUTE_HINTS) do
local value = instance:GetAttribute(name)
if value ~= nil and truthyPaidValue(value) then
return true
end
end
for name, value in pairs(instance:GetAttributes()) do
local lower = string.lower(tostring(name))
if (
lower:find("product", 1, true)
or lower:find("gamepass", 1, true)
or lower:find("robux", 1, true)
or lower:find("premium", 1, true)
) and truthyPaidValue(value) then
return true
end
end
return false
end
local function textLooksPaid(value)
if type(value) ~= "string" or value == "" then
return false
end
local lower = string.lower(value)
return lower:find("robux", 1, true) ~= nil
or lower:find("r$", 1, true) ~= nil
or lower:find("gamepass", 1, true) ~= nil
or lower:find("game pass", 1, true) ~= nil
or lower:find("developer product", 1, true) ~= nil
or lower:find("premium", 1, true) ~= nil
or lower:find("buy with robux", 1, true) ~= nil
end
local function nameLooksPaid(name)
local lower = string.lower(tostring(name or ""))
return lower:find("robux", 1, true) ~= nil
or lower:find("gamepass", 1, true) ~= nil
or lower:find("game pass", 1, true) ~= nil
or lower:find("productid", 1, true) ~= nil
or lower:find("skipproduct", 1, true) ~= nil
or lower:find("premium", 1, true) ~= nil
end
local function isPaidBuildPad(pad)
if not pad or not pad.Parent then
return true
end
local objects = {pad}
for _, descendant in ipairs(pad:GetDescendants()) do
objects[#objects + 1] = descendant
end
for _, object in ipairs(objects) do
if hasPaidAttribute(object) or nameLooksPaid(object.Name) then
return true
end
if object:IsA("TextLabel")
or object:IsA("TextButton")
or object:IsA("TextBox")
then
if textLooksPaid(object.Text) then
return true
end
elseif object:IsA("StringValue") then
if textLooksPaid(object.Value) then
return true
end
elseif object:IsA("IntValue") or object:IsA("NumberValue") then
if nameLooksPaid(object.Name) and object.Value > 0 then
return true
end
end
end
return false
end
