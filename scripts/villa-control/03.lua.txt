l distance = (root.Position - pos).Magnitude
if distance <= maxDistance and distance < nearestDistance then
nearest = instance
nearestDistance = distance
end
end
end
end
return nearest, nearestDistance
end

local TUNE_FIELDS = {
"PeakTorque",
"IdleTorque",
"RedlineTorque",
"T_Boost",
"ThrotAccel",
"SpeedLimit",
}
local function captureTune(tune, module)
if State.OriginalTunes[module] then
return State.OriginalTunes[module]
end
local original = {}
for _, key in ipairs(TUNE_FIELDS) do
original[key] = tune[key]
end
State.OriginalTunes[module] = original
return original
end
local function applyTune(showNotice)
local tune, module = currentTune()
if not tune or not module then
if showNotice then
notify("Vehicle Tune", "Sit in an A-Chassis vehicle first.", "car")
end
return false
end
local original = captureTune(tune, module)
if type(original.PeakTorque) == "number" then
tune.PeakTorque = original.PeakTorque * State.Tune.TorqueMult
end
if type(original.IdleTorque) == "number" then
tune.IdleTorque = original.IdleTorque * State.Tune.TorqueMult
end
if type(original.RedlineTorque) == "number" then
tune.RedlineTorque = original.RedlineTorque * State.Tune.TorqueMult
end
if type(original.T_Boost) == "number" then
tune.T_Boost = original.T_Boost * State.Tune.TurboMult
end
if type(original.ThrotAccel) == "number" then
tune.ThrotAccel = original.ThrotAccel * State.Tune.ThrottleMult
end
if type(original.SpeedLimit) == "number" then
tune.SpeedLimit = State.Tune.SpeedLimit
end
if showNotice then
notify(
"Vehicle Tune",
("Applied: torque x%.2f | turbo x%.2f | throttle x%.2f | limit %s")
:format(
State.Tune.TorqueMult,
State.Tune.TurboMult,
State.Tune.ThrottleMult,
State.Tune.SpeedLimit == 0 and "unlimited" or tostring(State.Tune.SpeedLimit)
),
"gauge"
)
end
return true
end
local function resetTune()
local tune, module = currentTune()
if not tune or not module then
notify("Vehicle Tune", "Sit in the tuned vehicle first.", "car")
return
end
local original = State.OriginalTunes[module]
if not original then
notify("Vehicle Tune", "No saved original tune for this vehicle.", "rotate-ccw")
return
end
for key, value in pairs(original) do
tune[key] = value
end
State.OriginalTunes[module] = nil
State.Tune.TorqueMult = 1
State.Tune.TurboMult = 1
State.Tune.ThrottleMult = 1
State.Tune.SpeedLimit = 0
notify("Vehicle Tune", "Original A-Chassis tune restored.", "rotate-ccw")
end

local function fireLights(...)
local fe = lightRemote()
if not fe then
notify("Vehicle Lights", "FE_Lights remote was not found on your car.", "lightbulb-off")

end

local ok = pcall(function()
fe:FireServer(table.unpack(args, 1, args.n))
end)
return ok
end
