    if not player then
        return nil
    end


    return GetRoundData()[
        player.Name
    ]
end


local function GetRole(
    player
)

    local data =
        PlayerData(player)


    if not data then
        return "Unknown"
    end


    return tostring(
        data.Role
        or "Unknown"
    )
end


local function IsAlive(
    player
)

    local data =
        PlayerData(player)


    if data
        and data.Dead == true
    then

        return false
    end


    local character =
        player
        and player.Character


    local humanoid =
        character
        and character:
            FindFirstChildOfClass(
                "Humanoid"
            )


    if humanoid then
        return humanoid.Health > 0
    end


    return data
        and data.Dead == false
        or false
end


--==============================================================
-- ROLE CACHE
--==============================================================

local function UpdateRoleCache()

    local murderer = nil

    local sheriff = nil

    local hero = nil


    local data =
        GetRoundData()


    for name,
        info
    in pairs(data)
    do

        local player =
            Players:
            FindFirstChild(name)


        if player
            and info.Dead ~= true
        then

            if info.Role == "Murderer" then

                murderer = player

            elseif info.Role == "Sheriff" then

                sheriff = player

            elseif info.Role == "Hero" then

                hero = player
            end
        end
    end


    Runtime.Murderer =
        murderer

    Runtime.Sheriff =
        sheriff

    Runtime.Hero =
        hero

    Runtime.LocalRole =
        GetRole(
            LocalPlayer
        )

    Runtime.RoundActive =
        next(data) ~= nil

    Runtime.LastRoleRefresh =
        os.clock()
end

--==============================================================
-- CHARACTER TARGET
--==============================================================

local function TargetPart(
    player
)

    local character =
        player
        and player.Character


    if not character then
        return nil
    end


    return character:
        FindFirstChild(
            "HumanoidRootPart"
        )

        or character:
            FindFirstChild(
                "UpperTorso"
            )

        or character:
            FindFirstChild(
                "Torso"
            )

        or character:
            FindFirstChild(
                "Head"
            )
end


local function DistanceTo(
    player
)

    local _,
        _,
        root =
        GetCharacter()


    local target =
        TargetPart(player)


    if not root
        or not target
    then

        return math.huge
    end


    return (
        root.Position
        - target.Position
    ).Magnitude
end

end

local function DestroyPlayerESP(
    player
)

    local esp =
        Runtime.PlayerESP[
            player
        ]


    if not esp then
        return
    end


    pcall(function()

        if esp.Highlight then
            esp.Highlight:Destroy()
        end


        if esp.Billboard then
            esp.Billboard:Destroy()
        end
    end)


    Runtime.PlayerESP[
        player
    ] =
        nil
end


local function ClearPlayerESP()

    local copy = {}

    for player
    in pairs(
        Runtime.PlayerESP
    )
    do

        copy[#copy + 1] =
            player
    end


    for _,
        player
    in ipairs(copy)
    do

        DestroyPlayerESP(
            player
        )
    end
end
