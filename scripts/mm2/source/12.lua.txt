local function SetPerformanceMode(
    enabled
)

    Settings.PerformanceMode =
        enabled


    if enabled then

        Settings.ESPInterval =
            0.45


        Settings.GunESPInterval =
            1.00


        Settings.CoinESPInterval =
            1.75


        Settings.MaxCoinESP =
            18


        Settings.WarningInterval =
            0.28


        Notify(
            "Performance Mode",
            "ON • lowest background load 🧊",
            "cpu"
        )

    else

        Settings.ESPInterval =
            0.28


        Settings.GunESPInterval =
            0.55


        Settings.CoinESPInterval =
            1.0


        Settings.MaxCoinESP =
            32


        Settings.WarningInterval =
            0.15


        Notify(
            "Performance Mode",
            "OFF • faster visual updates",
            "gauge"
        )
    end
end

local function CountTable(
    t
)

    local count = 0


    for _ in pairs(t) do
        count += 1
    end


    return count
end
