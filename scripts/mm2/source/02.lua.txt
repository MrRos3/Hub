--==============================================================
-- NOTIFY
--==============================================================

local function Notify(
    title,
    text,
    icon
)

    print(
        "[Salty MM2]",
        title,
        text
    )


    if not Runtime.VantaUI then
        return
    end


    pcall(function()

        Runtime.VantaUI:
        Notify({

            Title = tostring(title),

            Content = tostring(text),

            Desc = tostring(text),

            Icon = icon or "info",

            Duration = 3,
        })

    end)
end


--==============================================================
-- CHARACTER
--==============================================================

local function GetCharacter()

    local character =
        LocalPlayer.Character


    if not character then
        return nil, nil, nil
    end


    return
        character,

        character:
            FindFirstChildOfClass(
                "Humanoid"
            ),

        character:
            FindFirstChild(
                "HumanoidRootPart"
            )
end


local function CharacterAlive()

    local _,
        humanoid =
        GetCharacter()


    return humanoid
        and humanoid.Health > 0
end


local function StopMomentum()

    local _,
        _,
        root =
        GetCharacter()


    if root then

        root.AssemblyLinearVelocity =
            Vector3.zero

        root.AssemblyAngularVelocity =
            Vector3.zero
    end
end


--==============================================================
-- ROUND DATA
--==============================================================

local function GetRoundData()

    if type(
        CurrentRoundClient.PlayerData
    ) == "table"
    then

        return CurrentRoundClient.PlayerData
    end


    return {}
end


local function RefreshRoundData()

    if type(
        CurrentRoundClient
            .GetLatestPlayerData
    ) ~= "function"
    then

        return false
    end


    local ok,
        data =
        pcall(function()

            return CurrentRoundClient
                .GetLatestPlayerData()

        end)


    if not ok then

        ok,
        data =
            pcall(function()

                return CurrentRoundClient:
                    GetLatestPlayerData()

            end)
    end


    if ok
        and type(data)
            == "table"
    then

        CurrentRoundClient.PlayerData =
            data

        return true
    end


    return false
end
